# Requirements for Constrained Decoding Example Models
# Used by: scripts/models/*-example
#
# These dependencies enable structured output generation from LLMs using
# various constrained decoding libraries. Each library uses a different
# approach to ensure valid output:
#
# - Outlines: Token masking via JSON Schema/regex/grammar
# - Guidance: Template DSL with interleaved generation
# - LMQL: SQL-like query language for constraints
# - Jsonformer: Structural token generation (JSON only)
# - Instructor: Pydantic validation with retry logic
#
# Install: pip install -r docker/requirements-constrained-decoding.txt

# =============================================================================
# Core ML Dependencies
# =============================================================================

# PyTorch - Base ML framework
torch>=2.0.0

# HuggingFace Transformers - Model loading and inference
transformers>=4.35.0
tokenizers>=0.14.0

# =============================================================================
# Constrained Decoding Libraries
# =============================================================================

# Outlines - Token-level constrained generation
# Supports: JSON Schema, regex, context-free grammars
# https://github.com/outlines-dev/outlines
outlines>=0.0.37

# Guidance - Template-based constrained generation
# Supports: Template DSL with gen/select blocks, control flow
# https://github.com/guidance-ai/guidance
guidance>=0.1.10

# LMQL - Query language for LLMs
# Supports: SQL-like constraints, type checking, scripted prompts
# https://github.com/eth-sri/lmql
lmql>=0.7.0

# Jsonformer - Lightweight JSON generation
# Supports: JSON Schema, structural generation
# https://github.com/1rgs/jsonformer
jsonformer>=0.12.0

# Instructor - Pydantic-based structured output
# Supports: Pydantic validation, automatic retry, OpenAI API
# https://github.com/jxnl/instructor
instructor>=1.0.0

# =============================================================================
# Supporting Libraries
# =============================================================================

# Pydantic - Data validation (used by Outlines, Instructor)
pydantic>=2.0.0

# OpenAI client (optional, for Instructor with API models)
openai>=1.0.0

# Interegular - Regex to FSM conversion (used by Outlines)
interegular>=0.3.0

# =============================================================================
# Optional: Additional backends
# =============================================================================
